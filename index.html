import "jsr:@supabase/functions-js/edge-runtime.d.ts"; import { createClient } from "npm:@supabase/supabase-js@2"; const SUPABASE_URL = Deno.env.get("SUPABASE_URL")!; const TELEGRAM_BOT_TOKEN = Deno.env.get("TELEGRAM_BOT_TOKEN")!; /* ============================== HMAC-SHA256 ============================== */ async function hmacSha256( keyData: Uint8Array, message: string ): Promise { const key = await crypto.subtle.importKey( "raw", keyData, { name: "HMAC", hash: "SHA-256" }, false, ["sign"] ); } const userText = params.get("user"); if (!userText) { throw new Error( "Telegram user missing" ); } let user; try { user = JSON.parse(userText); } catch { throw new Error( "Telegram user data invalid" ); } if (!user.id) { throw new Error( "Telegram user ID missing" ); } return user; } /* ============================== INDIA DATE ============================== */ function indiaDate(): string { const parts = new Intl.DateTimeFormat( "en-CA", { timeZone: "Asia/Kolkata", year: "numeric", month: "2-digit", day: "2-digit" } ).formatToParts(new Date()); const year = parts.find( (x) => x.type === "year" )?.value; const month = parts.find( (x) => x.type === "month" )?.value; const day = parts.find( (x) => x.type === "day" )?.value; return `${year}-${month}-${day}`; } /* ============================== YESTERDAY ============================== */ function previousDate( dateString: string ): string { const d = new Date( `${dateString}T00:00:00+05:30` ); d.setUTCDate( d.getUTCDate() - 1 );
